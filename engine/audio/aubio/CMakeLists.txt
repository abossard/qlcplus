# Minimal CMake build for aubio. Builds a static library from an external
# aubio source tree (no bundled copy in this repository) without dragging in
# external dependencies (sndfile / avcodec / FLAC / vorbis / rubberband / fftw).
#
# Point AUBIO_SOURCE_DIR at the `src/` directory of an aubio checkout
# (compatible with 0.4.9 and 0.5.0-alpha layouts):
#   cmake -DAUBIO_SOURCE_DIR=/path/to/aubio/src ...
# When it is empty, the pinned revision below is downloaded instead.
#
# We deliberately do NOT call upstream's root or src/ CMakeLists.txt — it is
# SHARED-only, hard-requires PkgConfig, lists I/O and effects sources
# unconditionally, and does not wire Accelerate/FFTW config correctly. We
# treat the external repository as a source tree only. SOURCE_SUBDIR therefore
# points at a directory that does not exist, which is how FetchContent is told
# to download without adding the dependency's own project to the build.

cmake_minimum_required(VERSION 3.18)

set(AUBIO_SOURCE_DIR ""
    CACHE PATH "Path to aubio source tree (the src/ subdir of an aubio checkout). Empty: download AUBIO_GIT_TAG.")
set(AUBIO_GIT_TAG "ad5cf975aed08cc4562dd008cf9f83b12b82ffb8"
    CACHE STRING "aubio revision downloaded when AUBIO_SOURCE_DIR is empty")

if(NOT AUBIO_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(aubio
        GIT_REPOSITORY https://github.com/aubio/aubio.git
        GIT_TAG ${AUBIO_GIT_TAG}
        SOURCE_SUBDIR qlcplus-does-not-configure-aubio
    )
    FetchContent_MakeAvailable(aubio)
    set(AUBIO_SOURCE_DIR "${aubio_SOURCE_DIR}/src")
endif()

if(NOT EXISTS "${AUBIO_SOURCE_DIR}/aubio.h")
    message(FATAL_ERROR
        "AUBIO_SOURCE_DIR (${AUBIO_SOURCE_DIR}) must point to an aubio source "
        "tree containing aubio.h. Set -DAUBIO_SOURCE_DIR=/path/to/aubio/src.")
endif()

# Collect all .c source files from aubio's src tree.
# CONFIGURE_DEPENDS so adding/removing files in the external tree triggers a
# re-glob on the next build.
file(GLOB_RECURSE AUBIO_SOURCES CONFIGURE_DEPENDS
    "${AUBIO_SOURCE_DIR}/*.c"
)

# Exclude I/O sources (audio capture/decoding is handled by QLC+).
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/io/")
# Exclude synthesis sources (sampler/wavetable depend on parameter+resampler chains
# we don't pull in).
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/synth/")
# Exclude effects/ (rubberband-backed pitch shift / time stretch — needs librubberband).
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/effects/")
# Exclude FFT/DCT backends that need external libs we don't link against.
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/dct_fftw\\.c$")
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/dct_ipp\\.c$")
# Exclude Windows-only DLL entry point (lives in utils/ in 0.5.x).
list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/windll\\.c$")

# Pick the DCT backend matching the FFT backend defined below.
# Keep dct_plain.c on every platform — dct.c falls back to it.
if(APPLE)
    list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/dct_ooura\\.c$")
else()
    list(FILTER AUBIO_SOURCES EXCLUDE REGEX "/dct_accelerate\\.c$")
endif()

# Generate a tiny include shim so QLC+ code can keep using
# `#include <aubio/aubio.h>` (matching aubio's installed include layout)
# without symlinks. The real aubio.h lives at ${AUBIO_SOURCE_DIR}/aubio.h;
# its quoted internal includes ("types.h", "spectral/fft.h", ...) resolve
# relative to that file, so consumers do not need any extra include path.
set(AUBIO_INCLUDE_SHIM_DIR "${CMAKE_CURRENT_BINARY_DIR}/aubio_include")
file(MAKE_DIRECTORY "${AUBIO_INCLUDE_SHIM_DIR}/aubio")
file(WRITE "${AUBIO_INCLUDE_SHIM_DIR}/aubio/aubio.h"
"/* Auto-generated by qlcplus — forwards <aubio/aubio.h> to the external aubio source tree. */
#pragma once
#include \"${AUBIO_SOURCE_DIR}/aubio.h\"
")

add_library(qlcplusaubio STATIC ${AUBIO_SOURCES})
set_property(TARGET qlcplusaubio PROPERTY POSITION_INDEPENDENT_CODE ON)

# Public include dir: the generated shim, so consumers write `#include <aubio/aubio.h>`.
# Private include dir: AUBIO_SOURCE_DIR, so internal aubio sources find
# "aubio_priv.h", "fvec.h", "spectral/fft.h", etc.
target_include_directories(qlcplusaubio
    PUBLIC  "${AUBIO_INCLUDE_SHIM_DIR}"
    PRIVATE "${AUBIO_SOURCE_DIR}"
)

# Feature defines. We rely on aubio's built-in Ooura FFT (or Accelerate on Apple),
# so no external FFT/audio dependency is needed.
target_compile_definitions(qlcplusaubio PRIVATE
    HAVE_STDLIB_H=1
    HAVE_STDIO_H=1
    HAVE_MATH_H=1
    HAVE_STRING_H=1
    HAVE_LIMITS_H=1
    HAVE_STDARG_H=1
    HAVE_ERRNO_H=1
    HAVE_C99_VARARGS_MACROS=1
    AUBIO_LOG_DOMAIN="qlcplus"
)

if(APPLE)
    target_compile_definitions(qlcplusaubio PRIVATE HAVE_ACCELERATE=1)
    target_link_libraries(qlcplusaubio PRIVATE "-framework Accelerate")
endif()

if(WIN32)
    target_compile_definitions(qlcplusaubio PRIVATE HAVE_WIN_HACKS=1)
endif()

# Vendored third-party code: silence all warnings to keep parent -Werror builds clean.
if(MSVC)
    target_compile_options(qlcplusaubio PRIVATE /w)
else()
    target_compile_options(qlcplusaubio PRIVATE -w)
endif()
